<body>

    <div id="heart-container"></div>

    <script>
        // Данные для большого сердца (фаза удара)
        const BIG = [
            "  **********      **********  ",
            "**************  **************",
            "********************************",
            "********************************",
            "********************************",
            "  ****************************  ",
            "    ************************    ",
            "      **********************    ",
            "        ******************      ",
            "          **************        ",
            "            **********          ",
            "              ******            ",
            "                **              ",
        ];

        // Данные для маленького сердца (фаза расслабления)
        const SMALL = [
            "  ********    ********  ",
            "************  **********",
            "************************",
            "************************",
            "  **********************",
            "    ******************  ",
            "      **************    ",
            "        **********      ",
            "          ******        ",
            "            **          ",
        ];

        const container = document.getElementById('heart-container');
        let builtLines = ;
        
        // Настройки скорости (в миллисекундах)
        const buildDelay = 150;   // Скорость сборки (быстрее, чем в консольном варианте)
        const pulseDelay = 300;   // Скорость пульсации
        
        let isBuilding = true;

        // Функция отрисовки: превращает массив строк в HTML
        function render() {
            container.innerHTML = builtLines.join('<br>');
        }

        // Этап 1: Сборка сердца по строчкам
        function startBuild() {
            if (isBuilding) {
                if (builtLines.length < BIG.length) {
                    // Добавляем следующую строку из большого сердца
                    builtLines.push(BIG[builtLines.length]);
                    render();
                    // Планируем следующий шаг
                    setTimeout(startBuild, buildDelay);
                } else {
                    // Сердце собрано, переходим к пульсации
                    isBuilding = false;
                    startPulse();
                }
            }
        }

        // Этап 2: Бесконечная пульсация
        function startPulse() {
            // Фаза "Удар" (большое сердце)
            builtLines = [...BIG];
            render();
            
            setTimeout(() => {
                // Фаза "Расслабление" (маленькое сердце)
                builtLines = [...SMALL];
                render();
                
                // Повторяем цикл пульсации
                setTimeout(startPulse, pulseDelay);
            }, pulseDelay);
        }

        // Запуск анимации
        startBuild();
    </script>
